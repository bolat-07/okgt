function hav(a,b,c,d){
  var R=6371,dl=(c-a)*Math.PI/180,dn=(d-b)*Math.PI/180;
  var x=Math.sin(dl/2)*Math.sin(dl/2)+Math.cos(a*Math.PI/180)*Math.cos(c*Math.PI/180)*Math.sin(dn/2)*Math.sin(dn/2);
  return R*2*Math.asin(Math.sqrt(x));
}
function interp(pts,km){
  var cum=0;
  for(var i=1;i<pts.length;i++){
    var seg=hav(pts[i-1].lat,pts[i-1].lng,pts[i].lat,pts[i].lng);
    if(cum+seg>=km){
      var t=(km-cum)/seg;
      return{lat:pts[i-1].lat+t*(pts[i].lat-pts[i-1].lat),lng:pts[i-1].lng+t*(pts[i].lng-pts[i-1].lng),prev:pts[i-1],next:pts[i],dp:km-cum,dn:seg-(km-cum)};
    }
    cum+=seg;
  }
  var L=pts[pts.length-1];return{lat:L.lat,lng:L.lng,prev:L,next:L,dp:0,dn:0};
}

  var dist=parseFloat(document.getElementById('distInp').value);
  if(!code||isNaN(dist)||dist<0){toast('⚠️ Заполните все поля');return;}
  var opt=ALL_OPTIONS[code];
  if(!opt){toast('⚠️ Линия не найдена');return;}
  if(dist>opt.total_km){toast('⚠️ Расстояние '+dist+' км > длины участка '+opt.total_km+' км');return;}
  var btn=document.getElementById('goBtn');
  btn.disabled=true;btn.textContent='⏳ Обработка...';
  document.getElementById('icard').classList.remove('show');
  document.getElementById('rcard').classList.remove('show');
  setStatus('ld','🔍','Вычисляю место повреждения...','');
  var effectiveDist=currentDir==='B'?opt.total_km-dist:dist;
  var pt=interp(opt.points,effectiveDist);
  var fLat=pt.lat,fLng=pt.lng;
  currentFaultLat=fLat;currentFaultLng=fLng;
  if(faultM)map.removeLayer(faultM);
  var fi=L.divIcon({className:'',html:'<div class="fp"></div>',iconSize:[22,22],iconAnchor:[11,11]});
  faultM=L.marker([fLat,fLng],{icon:fi,zIndexOffset:1000}).addTo(map)
    .bindPopup('<b style="color:#ff4757">⚡ ПОВРЕЖДЕНИЕ</b><br><span style="color:#fff">'+opt.name+'</span><br><span style="color:rgba(255,255,255,.6)">'+dist.toFixed(2)+' км</span><br><span style="color:rgba(255,255,255,.45);font-size:11px">'+fLat.toFixed(5)+', '+fLng.toFixed(5)+'</span>');
  var closer=pt.dp<=pt.dn?pt.prev:pt.next;
  var dirLabel=currentDir==='A'?'от начала':'от конца';
  document.getElementById('iLine').textContent=opt.name;
  document.getElementById('iDist').textContent=dist.toFixed(2)+' км '+dirLabel;
  document.getElementById('iMP').textContent=closer.name;
  document.getElementById('iMPd').textContent=Math.min(pt.dp,pt.dn).toFixed(3)+' км';
  document.getElementById('iCoords').textContent=fLat.toFixed(5)+', '+fLng.toFixed(5);
  document.getElementById('icard').classList.add('show');
  setStatus('ld','📍','Определяю ваше местоположение...','Разрешите доступ к геолокации');
  if(!navigator.geolocation){
    setStatus('er','⚠️','Геолокация недоступна','Откройте файл по HTTPS, либо используйте кнопки Google Maps/Яндекс');
    goTab('map');map.setView([fLat,fLng],14);faultM.openPopup();
    btn.disabled=false;btn.textContent='🔍 Показать повреждение и построить маршрут';return;
  }
  navigator.geolocation.getCurrentPosition(function(pos){
    myLat=pos.coords.latitude;myLng=pos.coords.longitude;
    if(myM)map.removeLayer(myM);
    var mic=L.divIcon({className:'',html:'<div class="my-dot"></div>',iconSize:[16,16],iconAnchor:[8,8]});
    myM=L.marker([myLat,myLng],{icon:mic}).addTo(map);
    setStatus('ok','📍','Местоположение найдено',myLat.toFixed(4)+', '+myLng.toFixed(4));
    buildRoute(myLat,myLng,fLat,fLng,btn);
  },function(e){
    setStatus('er','⚠️','Геолокация недоступна','Используйте кнопки Google Maps / Яндекс ниже');
    goTab('map');map.setView([fLat,fLng],14);faultM.openPopup();
    btn.disabled=false;btn.textContent='🔍 Показать повреждение и построить маршрут';
  },{enableHighAccuracy:true,timeout:12000,maximumAge:0});
}
function buildRoute(fromLat,fromLng,fLat,fLng,btn){
  setStatus('ld','🗺','Строю маршрут по дорогам...','');
  var url='https://router.project-osrm.org/route/v1/driving/'+fromLng+','+fromLat+';'+fLng+','+fLat+'?overview=full&geometries=geojson&steps=true';
  fetch(url)
    .then(function(r){return r.json();})
    .then(function(d){
      if(d.code!=='Ok')throw new Error('no route');
      var route=d.routes[0];
      var km=(route.distance/1000).toFixed(1);
      var mins=Math.round(route.duration/60);
      var h=Math.floor(mins/60),m=mins%60;
      if(routePoly)map.removeLayer(routePoly);
      routePoly=L.geoJSON(route.geometry,{style:{color:'#00e676',weight:5,opacity:.9}}).addTo(map);
      document.getElementById('rKm').textContent=km;
      document.getElementById('rMin').textContent=h>0?h+'ч '+m:String(mins);
      var sl=document.getElementById('slist');sl.innerHTML='';
      route.legs[0].steps.slice(0,10).forEach(function(s){
        var ic=sIcon(s.maneuver.type,s.maneuver.modifier);
        var dd=s.distance>=1000?(s.distance/1000).toFixed(1)+' км':Math.round(s.distance)+' м';
        sl.innerHTML+='<div class="si"><div class="sic">'+ic+'</div><div><div class="st">'+(s.name||sLbl(s.maneuver.type,s.maneuver.modifier))+'</div><div class="sd">'+dd+'</div></div></div>';
      });
      document.getElementById('rcard').classList.add('show');
      setStatus('ok','✅','Маршрут построен: '+km+' км',(h>0?h+' ч '+m:mins)+' мин в пути');
      map.fitBounds(routePoly.getBounds(),{padding:[50,50]});
      setTimeout(function(){faultM.openPopup();},600);
      goTab('map');
      toast('✅ '+km+' км · '+(h>0?h+'ч '+m:''+mins)+' мин');
    })
    .catch(function(){
      setStatus('er','⚠️','Нет дорог рядом','Показана точка — используйте координаты или кнопки навигаторов');
      goTab('map');map.setView([fLat,fLng],14);faultM.openPopup();
      toast('⚠️ Маршрут не найден — показана точка');
    })
    .finally(function(){btn.disabled=false;btn.textContent='🔍 Показать повреждение и построить маршрут';});
}
function sIcon(t,m){
  if(t==='arrive')return'🏁';if(t==='depart')return'🚀';if(t==='roundabout')return'🔄';
  if(m==='left')return'⬅️';if(m==='right')return'➡️';
  if(m==='slight left')return'↖️';if(m==='slight right')return'↗️';
  if(m==='uturn')return'🔃';return'⬆️';
}
function sLbl(t,m){
  if(t==='depart')return'Начало маршрута';if(t==='arrive')return'Пункт назначения';
  if(m==='left')return'Повернуть налево';if(m==='right')return'Повернуть направо';
  if(m==='straight')return'Ехать прямо';return'Продолжить движение';
}
var tt;
function toast(msg){
  var t=document.getElementById('toast');t.textContent=msg;t.classList.add('show');
  clearTimeout(tt);tt=setTimeout(function(){t.classList.remove('show');},3000);
}
